-- ── FPS BOOST ─────────────────────────────────────────────────────────────────
local origShadows = lighting.GlobalShadows

local function enableFPSBoost()
    pcall(function() settings().Rendering.QualityLevel = 1 end)
    lighting.GlobalShadows = false
    lighting.FogEnd = 9e9; lighting.FogStart = 9e9
    lighting.Brightness = 2
    lighting.EnvironmentDiffuseScale = 0
    lighting.EnvironmentSpecularScale = 0
    local atm = lighting:FindFirstChildOfClass("Atmosphere")
    if atm then atm.Density=0; atm.Glare=0; atm.Haze=0 end
    for _, v in ipairs(lighting:GetChildren()) do
        pcall(function()
            if v:IsA("BloomEffect") or v:IsA("ColorCorrectionEffect")
            or v:IsA("DepthOfFieldEffect") or v:IsA("SunRaysEffect")
            or v:IsA("BlurEffect") then v.Enabled=false end
        end)
    end
    for _, v in ipairs(ws:GetDescendants()) do
        pcall(function()
            if v:IsA("ParticleEmitter") or v:IsA("Trail")
            or v:IsA("Smoke") or v:IsA("Fire")
            or v:IsA("Sparkles") or v:IsA("Beam") then v.Enabled=false end
            if v:IsA("Decal") or v:IsA("Texture") then v.Transparency=1 end
            if v:IsA("BasePart") and not v:IsA("Terrain") then v.CastShadow=false end
        end)
    end
    pcall(function()
        ws.Terrain.WaterWaveSize=0; ws.Terrain.WaterWaveSpeed=0
        ws.Terrain.WaterReflectance=0; ws.Terrain.WaterTransparency=1
    end)
end

local function disableFPSBoost()
    pcall(function() settings().Rendering.QualityLevel=Enum.QualityLevel.Automatic end)
    lighting.GlobalShadows = origShadows
    for _, v in ipairs(lighting:GetChildren()) do
        pcall(function()
            if v:IsA("BloomEffect") or v:IsA("ColorCorrectionEffect")
            or v:IsA("DepthOfFieldEffect") or v:IsA("SunRaysEffect")
            or v:IsA("BlurEffect") then v.Enabled=true end
        end)
    end
end

-- ── HELPERS ───────────────────────────────────────────────────────────────────
local function getHum(p)     return p.Character and p.Character:FindFirstChildOfClass("Humanoid") end
local function getRoot(p)    return p.Character and p.Character:FindFirstChild("HumanoidRootPart") end
local function getHitPart(p) return p.Character and p.Character:FindFirstChild(S.HitPart) end
local function sameTeam(p)   return S.TeamCheck and plr.Team and p.Team and plr.Team==p.Team end

local function wallCheck(from,target)
    if not S.WallCheck then return true end
    local params=RaycastParams.new()
    params.FilterDescendantsInstances={plr.Character,target.Parent}
    params.FilterType=Enum.RaycastFilterType.Exclude
    local result=ws:Raycast(from,target.Position-from,params)
    return not result or result.Instance:IsDescendantOf(target.Parent)
end

local function getBest()
    local center=Vector2.new(cam.ViewportSize.X/2,cam.ViewportSize.Y/2)
    local best,bd=nil,S.FOV
    local myRoot=getRoot(plr)
    for _,p in ipairs(Players:GetPlayers()) do
        if p==plr or sameTeam(p) then continue end
        local hum=getHum(p)
        if not hum or hum.Health<=0 then continue end
        local part=getHitPart(p)
        if not part then continue end
        local sp,onScreen=cam:WorldToViewportPoint(part.Position)
        if not onScreen then continue end
        local d2=(Vector2.new(sp.X,sp.Y)-center).Magnitude
        if d2>S.FOV then continue end
        if myRoot and not wallCheck(myRoot.Position,part) then continue end
        if d2<bd then bd=d2; best=part end
    end
    return best
end

local espObjects = {}
-- espObjects[player] = { hl = Highlight, bb = BillboardGui, lbl = TextLabel }

local function cleanESP(p)
    if espObjects[p] then
        pcall(function() espObjects[p].hl:Destroy() end)
        pcall(function() espObjects[p].bb:Destroy() end)
        espObjects[p] = nil
    end
end

local function buildESPForPlayer(p)
    cleanESP(p)
    local char = p.Character; if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart"); if not root then return end

    local hl = Instance.new("Highlight")
    hl.FillColor         = Color3.fromRGB(255,40,40)
    hl.OutlineColor      = Color3.fromRGB(255,255,255)
    hl.FillTransparency  = 0.72
    hl.OutlineTransparency = 0
    hl.Adornee           = char
    hl.Parent            = ws

    local bb = Instance.new("BillboardGui")
    bb.Size         = UDim2.new(0,120,0,22)
    bb.StudsOffset  = Vector3.new(0,3.2,0)
    bb.AlwaysOnTop  = true
    bb.Adornee      = root
    bb.Parent       = ws

    local tLabel = Instance.new("TextLabel")
    tLabel.Size                    = UDim2.new(1,0,1,0)
    tLabel.BackgroundTransparency  = 1
    tLabel.TextColor3              = Color3.fromRGB(255,255,255)
    tLabel.Text                    = ""
    tLabel.Font                    = Enum.Font.GothamBold
    tLabel.TextSize                = 11
    tLabel.TextStrokeTransparency  = 0
    tLabel.Parent                  = bb

    espObjects[p] = { hl = hl, bb = bb, lbl = tLabel, root = root, char = char }
end

local function onPlayerRemoving(p)
    cleanESP(p)
end

local function onCharacterAdded(p, char)
    cleanESP(p)
    -- kurz warten bis HumanoidRootPart geladen
    task.delay(1, function()
        if S.ESPEnabled and p and p.Character == char then
            buildESPForPlayer(p)
        end
    end)
end

for _, p in ipairs(Players:GetPlayers()) do
    if p ~= plr then
        p.CharacterAdded:Connect(function(char) onCharacterAdded(p, char) end)
        p.CharacterRemoving:Connect(function() cleanESP(p) end)
    end
end

Players.PlayerAdded:Connect(function(p)
    p.CharacterAdded:Connect(function(char) onCharacterAdded(p, char) end)
    p.CharacterRemoving:Connect(function() cleanESP(p) end)
end)

Players.PlayerRemoving:Connect(onPlayerRemoving)

plr.CharacterAdded:Connect(function()
    -- eigenes Respawn: alle ESP neu bauen (Referenzen auf altes Char ungültig)
    for p, _ in pairs(espObjects) do cleanESP(p) end
    task.delay(1, function()
        if not S.ESPEnabled then return end
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= plr and p.Character then buildESPForPlayer(p) end
        end
    end)
end)

-- ── ESP UPDATE-LOOP (nur Werte schreiben, kein Destroy/Create) ────────────────
local lastESP = 0
RunService.Heartbeat:Connect(function()
    local t = tick()
    -- Update-Rate: 0.1s reicht für Labels (war 0.05s mit full rebuild)
    if t - lastESP < 0.1 then return end
    lastESP = t

    if not S.ESPEnabled then
        for p, obj in pairs(espObjects) do
            obj.hl.Enabled = false
            obj.bb.Enabled = false
        end
        return
    end

    local myRoot = getRoot(plr)

    for _, p in ipairs(Players:GetPlayers()) do
        if p == plr or sameTeam(p) then continue end

        local hum  = getHum(p)
        local obj  = espObjects[p]

        -- Spieler hat kein Objekt aber lebendes Char → bauen
        if not obj and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            buildESPForPlayer(p)
            obj = espObjects[p]
        end

        if not obj then continue end

        -- Prüfen ob Char noch gültig (Respawn-Lücke)
        if not obj.char or not obj.char.Parent then
            cleanESP(p)
            continue
        end

        local dead    = not hum or hum.Health <= 0
        local dist    = myRoot and (obj.root.Position - myRoot.Position).Magnitude or math.huge
        local inRange = dist <= S.MaxDist

        local show = (not dead) and inRange
        obj.hl.Enabled = show
        obj.bb.Enabled = show

        if show then
            local parts = {}
            if S.ShowName   then table.insert(parts, p.Name) end
            if S.ShowHealth then table.insert(parts, "HP:"..math.floor(hum.Health)) end
            if S.ShowDist   then table.insert(parts, math.floor(dist).."m") end
            obj.lbl.Text = table.concat(parts, " | ")
        end
    end
end)
